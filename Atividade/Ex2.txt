namespace Lista
{
    internal class Program
    {
        static void Main(string[] args)
        {
            List<double> P1 = new List<double> { 7.5, 8.0, 6.5, 9.0, 5.5 };
            List<double> P2 = new List<double> { 8.5, 7.0, 6.0, 9.5, 7.5 };

            int n = P1.Count;
            int m = P1.Count;

            double somaP1 = 0;
            foreach (double nota in P1)
                somaP1 += nota;
            double mediaP1 = somaP1 / n;

            double somaP2 = 0;
            foreach (double nota in P2)
                somaP2 += nota;
            double mediaP2 = somaP2 / m;

            Console.WriteLine($"Média da Prova 1: {mediaP1:F2}");
            Console.WriteLine($"Média da Prova 2: {mediaP2:F2}");

            if (mediaP1 > mediaP2)
                Console.WriteLine("A turma obteve melhor média na Prova 1.");
            else if (mediaP2 > mediaP1)
                Console.WriteLine("A turma obteve melhor média na Prova 2.");
            else
                Console.WriteLine("A turma obteve a mesma média nas duas provas.");
        }
    }
}
